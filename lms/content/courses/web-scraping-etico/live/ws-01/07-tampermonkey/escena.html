Tampermonkey: extrae mientras navegas

Un script que vive en tu navegador: lee la página que ya estás viendo, con tu sesión y a tu ritmo. Un clic, una página.

Ahora túSigue los cinco pasos y marca cada uno al terminarlo. Si no puedes instalar extensiones, usa la pestaña Sin instalar nada.

  1. Instala Tampermonkey en Chrome o Firefox desde su página oficial.
    https://www.tampermonkey.net/
    Si Chrome te pide permitir scripts de usuario, entra a chrome://extensions, abre «Detalles» de Tampermonkey y actívalo (o activa el modo de desarrollador).
  2. Ícono de Tampermonkey → Crear un nuevo script. Borra lo que trae, pega el script del recuadro «El script» y guarda con Ctrl+S (Cmd+S en Mac).
  3. Abre el sitio de práctica. Abajo a la derecha aparece tu cajita con el conteo.
    https://quotes.toscrape.com/
  4. Pasa a la página 2 y 3 con Next →. El conteo sube de 10 en 10. Recarga una página: no se repite nada.
  5. Pulsa CSV y JSON: se descargan frases.csv y frases.json. Ábrelos.

El script (cópialo completo) 
// ==UserScript==
// @name         Frases con permiso · JMD
// @namespace    jmd-learning
// @version      1.0
// @description  Junta las frases de quotes.toscrape.com mientras navegas y las exporta a CSV y JSON
// @match        https://quotes.toscrape.com/*
// @grant        none
// ==/UserScript==
(function () {
  'use strict';
  const CLAVE = 'jmd-frases';
  const limpiar = (t) => t.replace(/\s+/g, ' ').replace(/^[“"]|[”"]$/g, '').trim();
  const guardadas = JSON.parse(localStorage.getItem(CLAVE) || '[]');
  const vistas = new Set(guardadas.map((f) => f.frase + '|' + f.autor));
  let nuevas = 0;
  document.querySelectorAll('.quote').forEach((q) => {
    const fila = {
      frase: limpiar(q.querySelector('.text').textContent),
      autor: limpiar(q.querySelector('.author').textContent),
      etiquetas: [...q.querySelectorAll('.tag')].map((t) => t.textContent.trim()).join(';'),
      pagina: location.pathname,
    };
    const clave = fila.frase + '|' + fila.autor;
    if (!vistas.has(clave)) { vistas.add(clave); guardadas.push(fila); nuevas++; }
  });
  localStorage.setItem(CLAVE, JSON.stringify(guardadas));

  // Cajita abajo a la derecha: conteo y botones para exportar
  const caja = document.createElement('div');
  caja.style.cssText = 'position:fixed;right:12px;bottom:12px;z-index:9999;background:#14122b;color:#fff;' +
    'padding:10px 12px;border-radius:12px;font:14px system-ui;display:flex;gap:6px;align-items:center';
  const texto = document.createElement('span');
  texto.textContent = guardadas.length + ' frases (+' + nuevas + ' aquí)';
  caja.append(texto);
  const boton = (nombre, accion) => {
    const b = document.createElement('button');
    b.textContent = nombre; b.onclick = accion; caja.append(b);
  };
  const bajar = (archivo, tipo, contenido) => {
    const a = document.createElement('a');
    a.href = URL.createObjectURL(new Blob([contenido], { type: tipo }));
    a.download = archivo; a.click();
  };
  const csv = () => {
    const cols = ['frase', 'autor', 'etiquetas', 'pagina'];
    const celda = (v) => '"' + String(v).replace(/"/g, '""') + '"';
    return '\ufeff' + [cols.join(','), ...guardadas.map((f) => cols.map((c) => celda(f[c])).join(','))].join('\r\n');
  };
  boton('CSV', () => bajar('frases.csv', 'text/csv;charset=utf-8', csv()));
  boton('JSON', () => bajar('frases.json', 'application/json', JSON.stringify(guardadas, null, 2)));
  boton('Borrar', () => { localStorage.removeItem(CLAVE); location.reload(); });
  document.body.append(caja);
})();
