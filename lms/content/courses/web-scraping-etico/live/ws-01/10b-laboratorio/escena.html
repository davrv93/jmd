"""01 · Catálogo HTML con modales.

Recorre la Tienda JMD de práctica página por página (?page=N) siguiendo el enlace rel="next",
con pausa entre pedidos, y guarda catalogo.csv y catalogo.json.

Antes de pedir nada lee robots.txt: respeta Crawl-delay y se salta los enlaces prohibidos
(la tienda esconde uno, la «trampa» para bots que no leen las reglas).

    ALUMNO=tu-apodo CONTACTO=tu-correo@ejemplo.com python3 01_catalogo.py
"""
import csv
import json
import os
import random
import time
from urllib import robotparser
from urllib.parse import urljoin

import httpx
from bs4 import BeautifulSoup

LAB = os.environ.get("LAB", "https://jmd-learn.online/scraping-lab").rstrip("/")
ALUMNO = os.environ.get("ALUMNO", "sin-apodo")
CONTACTO = os.environ.get("CONTACTO", "sin-contacto")
PAUSA = float(os.environ.get("PAUSA", "1.0"))
# Identifícate: quién eres y cómo contactarte. El laboratorio lo muestra en el panel.
UA = f"KitJMD-catalogo/1.0 ({ALUMNO}; {CONTACTO})"

cliente = httpx.Client(headers={"User-Agent": UA, "X-Alumno": ALUMNO}, timeout=20, follow_redirects=True)


def pedir(url):
    """GET con reintentos: si el sitio dice 429 o 503, espera lo que pide Retry-After."""
    for intento in range(1, 6):
        r = cliente.get(url)
        if r.status_code in (429, 503):
            espera = int(r.headers.get("Retry-After", 2 ** intento))
            print(f"  {r.status_code}: el sitio pide esperar {espera} s (intento {intento})")
            time.sleep(espera)
            continue
        if r.status_code == 403:
            raise SystemExit(f"403 bloqueado: {r.text[:200]}… Espera y baja el ritmo.")
        r.raise_for_status()
        return r
    raise SystemExit(f"{url} no respondió tras 5 intentos")


# 1. Reglas primero
rp = robotparser.RobotFileParser()
rp.parse(pedir(LAB + "/robots.txt").text.splitlines())
PAUSA = max(PAUSA, float(rp.crawl_delay(UA) or 0))
inicio = LAB + "/tienda?page=1"
if not rp.can_fetch(UA, inicio):
    raise SystemExit("robots.txt no permite la tienda: me detengo")
print(f"robots.txt OK · pausa de {PAUSA} s entre pedidos · como {ALUMNO}")

# 2. Páginas: sigue rel="next" hasta que no haya más
filas, url, pagina, omitidos = [], inicio, 0, set()
while url:
    pagina += 1
    html = BeautifulSoup(pedir(url).text, "html.parser")
    for li in html.select("li.producto"):
        enlace = li.select_one("a.producto-enlace")
        filas.append({
            "id": int(li["data-id"]),
            "sku": li["data-sku"],
            "nombre": li.select_one(".nombre").get_text(strip=True),
            "categoria": li.select_one(".categoria").get_text(strip=True),
            "precio": float(li.select_one(".precio")["data-precio"]),
            "stock": int(li.select_one(".stock")["data-stock"]),
            "valoracion": float(li.select_one(".valoracion")["data-valor"]),
            "url": urljoin(url, enlace["href"]),
        })
    # Cualquier enlace que robots.txt prohíba se anota y NO se visita.
    for a in html.select("a[href]"):
        destino = urljoin(url, a["href"])
        if not rp.can_fetch(UA, destino):
            omitidos.add(destino)
    print(f"página {pagina}: {len(filas)} productos")
    sig = html.select_one('a[rel="next"]')
    url = urljoin(url, sig["href"]) if sig else None
    if url:
        time.sleep(PAUSA + random.uniform(0, 0.3))  # ritmo humano, con algo de azar

for o in sorted(omitidos):
    print(f"omitido por robots.txt: {o}")

# 3. Exportar: CSV que Excel abre con tildes (utf-8-sig) y JSON legible
with open("catalogo.csv", "w", newline="", encoding="utf-8-sig") as f:
    w = csv.DictWriter(f, fieldnames=list(filas[0].keys()))
    w.writeheader()
    w.writerows(filas)
with open("catalogo.json", "w", encoding="utf-8") as f:
    json.dump(filas, f, ensure_ascii=False, indent=2)
print(f"Listo: {len(filas)} productos en catalogo.csv y catalogo.json")

"""02 · API con sesión, cursor y «saltos» (reanudar donde quedaste).

1. Inicia sesión: pide el formulario, saca el token CSRF, hace POST con la misma cookie
   y lee el token de la API (cabecera X-Lab-Token).
2. Recorre /api/productos siguiendo next_cursor hasta que llegue null.
3. Guarda cada página en productos_api.jsonl y el cursor en checkpoint_api.json.
   Si se corta (Ctrl+C, red, bloqueo), al volver a correrlo SALTA lo ya bajado y sigue desde el cursor.

    ALUMNO=tu-apodo CONTACTO=tu-correo python3 02_api_sesion.py --parar-en 2   # simula un corte tras 2 páginas
    ALUMNO=tu-apodo CONTACTO=tu-correo python3 02_api_sesion.py                # reanuda y termina
    python3 02_api_sesion.py --desde-cero --q taza                              # borra el avance; filtra por texto
"""
import argparse
import json
import os
import time
from pathlib import Path

import httpx
from bs4 import BeautifulSoup

LAB = os.environ.get("LAB", "https://jmd-learn.online/scraping-lab").rstrip("/")
ALUMNO = os.environ.get("ALUMNO", "sin-apodo")
CONTACTO = os.environ.get("CONTACTO", "sin-contacto")
PAUSA = float(os.environ.get("PAUSA", "1.0"))
UA = f"KitJMD-api/1.0 ({ALUMNO}; {CONTACTO})"

SALIDA = Path("productos_api.jsonl")
CHECKPOINT = Path("checkpoint_api.json")

ap = argparse.ArgumentParser()
ap.add_argument("--q", default="", help="texto a buscar (vacío = todo el catálogo)")
ap.add_argument("--limite", type=int, default=20, help="productos por página (máx. 50)")
ap.add_argument("--parar-en", type=int, default=0, help="se corta a propósito tras N páginas")
ap.add_argument("--desde-cero", action="store_true", help="borra checkpoint y salida")
args = ap.parse_args()


def pedir(c, metodo, url, **kw):
    for intento in range(1, 6):
        r = c.request(metodo, url, **kw)
        if r.status_code in (429, 503):
            espera = int(r.headers.get("Retry-After", 2 ** intento))
            print(f"  {r.status_code}: espero {espera} s como pide Retry-After")
            time.sleep(espera)
            continue
        return r
    raise SystemExit("el sitio no responde: me detengo")


def iniciar_sesion():
    c = httpx.Client(base_url=LAB, headers={"User-Agent": UA, "X-Alumno": ALUMNO}, timeout=20, follow_redirects=True)
    form = BeautifulSoup(pedir(c, "GET", "/login").text, "html.parser")
    csrf = form.select_one('input[name="csrf_token"]')["value"]       # sale del HTML
    r = pedir(c, "POST", "/login", data={"csrf_token": csrf, "usuario": ALUMNO, "clave": "practica"})
    if r.status_code != 200:
        raise SystemExit(f"no se inició sesión ({r.status_code}): {r.text[:300]}")
    token = BeautifulSoup(r.text, "html.parser").select_one("#api-token").get_text(strip=True)
    c.headers["X-Lab-Token"] = token                                   # la cookie ya la guarda el cliente
    print(f"sesión iniciada como {ALUMNO} · cookie lab_sesion y token {token[:8]}… (no lo imprimas entero)")
    return c


if args.desde_cero:
    CHECKPOINT.unlink(missing_ok=True)
    SALIDA.unlink(missing_ok=True)

# Lo que ya tengo (para saltarlo) y dónde me quedé
ya = set()
if SALIDA.exists():
    for linea in SALIDA.read_text(encoding="utf-8").splitlines():
        if linea.strip():
            ya.add(json.loads(linea)["id"])
estado = json.loads(CHECKPOINT.read_text()) if CHECKPOINT.exists() else {}
if estado.get("q", "") != args.q:
    estado = {}                                     # otra búsqueda: el cursor viejo no sirve
if estado.get("terminado"):
    print(f"ya estaba completo: {len(ya)} productos en {SALIDA}. Usa --desde-cero para repetir.")
    raise SystemExit(0)
cursor = estado.get("cursor")
if cursor:
    print(f"reanudo desde el checkpoint: {len(ya)} productos ya bajados, salto directo al cursor guardado")

c = iniciar_sesion()
paginas = 0
while True:
    params = {"limite": args.limite, "q": args.q}
    if cursor:
        params["cursor"] = cursor
    r = pedir(c, "GET", "/api/productos", params=params)
    if r.status_code == 401:                         # sesión vencida: entro de nuevo y sigo con el mismo cursor
        print("  401: sesión vencida, vuelvo a entrar")
        c = iniciar_sesion()
        continue
    if r.status_code == 403:
        raise SystemExit(f"403 bloqueado: {r.json().get('motivo')} · vuelve a correrlo luego: reanudará solo")
    r.raise_for_status()
    datos = r.json()
    nuevos = [p for p in datos["items"] if p["id"] not in ya]
    with SALIDA.open("a", encoding="utf-8") as f:
        for p in nuevos:
            f.write(json.dumps(p, ensure_ascii=False) + "\n")
            ya.add(p["id"])
    cursor = datos["next_cursor"]
    # El checkpoint se escribe DESPUÉS de guardar los datos: si se corta en medio, a lo mucho repites una página.
    CHECKPOINT.write_text(json.dumps({"cursor": cursor, "q": args.q, "terminado": cursor is None, "productos": len(ya)}))
    paginas += 1
    print(f"página {paginas}: +{len(nuevos)} nuevos ({len(datos['items']) - len(nuevos)} saltados) · {len(ya)}/{datos['total']}")
    if cursor is None:
        break
    if args.parar_en and paginas >= args.parar_en:
        print(f"corte simulado tras {paginas} páginas. Vuelve a correr el script: seguirá desde el checkpoint.")
        raise SystemExit(3)
    time.sleep(PAUSA)

print(f"Listo: {len(ya)} productos en {SALIDA} · checkpoint marcado como terminado")

El script (por si no bajaste el kit)
$ cat > 03_descarga_cola.py <<'EOF'
"""03 · Descarga masiva con cola (asyncio).

- Lee la lista de 200 archivos de /api/descargas (nombre, url, tamaño y sha256).
- Mete en la cola solo los que FALTAN: si el archivo ya está y su sha256 cuadra, lo salta.
- N trabajadores sacan de la cola. Un «semáforo de ritmo» compartido reparte los pedidos para no pasar
  el límite del sitio, aunque haya muchos trabajadores.
- Reintenta 500/502/503/504 con espera exponencial (1, 2, 4, 8 s + azar) y respeta Retry-After en 429/503.
- Si llega un 403 (bloqueo), TODOS los trabajadores se detienen el tiempo que pide el sitio.
- Escribe en .part y renombra al final: un corte nunca deja un archivo a medias que parezca bueno.

    ALUMNO=tu-apodo CONTACTO=tu-correo python3 03_descarga_cola.py
    TRABAJADORES=8 python3 03_descarga_cola.py      # más trabajadores NO es más rápido: el ritmo lo pone el sitio
"""
import asyncio
import hashlib
import os
import random
import sys
import time
from pathlib import Path

import httpx

LAB = os.environ.get("LAB", "https://jmd-learn.online/scraping-lab").rstrip("/")
ALUMNO = os.environ.get("ALUMNO", "sin-apodo")
CONTACTO = os.environ.get("CONTACTO", "sin-contacto")
TRABAJADORES = int(os.environ.get("TRABAJADORES", "4"))
REINTENTOS = int(os.environ.get("REINTENTOS", "5"))
UA = f"KitJMD-cola/1.0 ({ALUMNO}; {CONTACTO})"
CARPETA = Path(os.environ.get("CARPETA", "descargas"))


class Ritmo:
    """Deja pasar un pedido cada `intervalo` segundos, compartido por todos los trabajadores."""

    def __init__(self, intervalo):
        self.intervalo = intervalo
        self.siguiente = 0.0
        self.candado = asyncio.Lock()
        self.pausa_hasta = 0.0          # la fija un 403 o un 429: frena a todos

    async def turno(self):
        async with self.candado:
            ahora = time.monotonic()
            espera = max(self.siguiente, self.pausa_hasta) - ahora
            if espera > 0:
                await asyncio.sleep(espera)
            self.siguiente = time.monotonic() + self.intervalo

    def frenar(self, segundos):
        self.pausa_hasta = max(self.pausa_hasta, time.monotonic() + segundos)


def sha256(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


class Barra:
    def __init__(self, total):
        self.total, self.ok, self.mal, self.reint, self.t0 = total, 0, 0, 0, time.monotonic()

    def pintar(self):
        hechos = self.ok + self.mal
        ancho = 30
        lleno = int(ancho * hechos / self.total) if self.total else ancho
        seg = time.monotonic() - self.t0
        sys.stderr.write(f"\r[{'#' * lleno}{'.' * (ancho - lleno)}] {hechos}/{self.total} · ok {self.ok} · fallidos {self.mal} · reintentos {self.reint} · {seg:5.1f} s ")
        sys.stderr.flush()


async def bajar(cliente, ritmo, arch, barra):
    destino = CARPETA / arch["nombre"]
    for intento in range(1, REINTENTOS + 1):
        await ritmo.turno()
        try:
            r = await cliente.get(arch["url"])
        except httpx.TransportError as e:
            r, error = None, str(e)
        if r is not None and r.status_code == 200:
            datos = r.content
            if hashlib.sha256(datos).hexdigest() != arch["sha256"]:
                error = "sha256 no cuadra"
            else:
                parcial = destino.with_suffix(".part")
                parcial.write_bytes(datos)
                parcial.replace(destino)          # atómico: o está entero o no está
                return True
        elif r is not None:
            error = f"HTTP {r.status_code}"
            if r.status_code in (429, 403) or (r.status_code == 503 and "Retry-After" in r.headers):
                espera = int(r.headers.get("Retry-After", "5"))
                ritmo.frenar(espera)              # el sitio manda: frena a TODOS
                if r.status_code == 403:
                    sys.stderr.write(f"\n403: el sitio nos bloqueó {espera} s. Todos esperan.\n")
                barra.reint += 1
                barra.pintar()
                continue
            if r.status_code not in (500, 502, 504):
                break                             # 404 y compañía: reintentar no sirve
        if intento == REINTENTOS:
            break
        espera = min(2 ** (intento - 1), 8) + random.uniform(0, 0.5)    # 1, 2, 4, 8 s + azar
        barra.reint += 1
        barra.pintar()
        await asyncio.sleep(espera)
    sys.stderr.write(f"\n  {arch['nombre']}: me rindo tras {intento} intentos ({error})\n")
    return False


async def trabajador(cola, cliente, ritmo, barra, fallidos):
    while True:
        arch = await cola.get()
        try:
            if await bajar(cliente, ritmo, arch, barra):
                barra.ok += 1
            else:
                barra.mal += 1
                fallidos.append(arch["nombre"])
            barra.pintar()
        finally:
            cola.task_done()


async def main():
    CARPETA.mkdir(exist_ok=True)
    cab = {"User-Agent": UA, "X-Alumno": ALUMNO}
    limites = httpx.Limits(max_connections=TRABAJADORES)
    async with httpx.AsyncClient(headers=cab, timeout=20, limits=limites) as cliente:
        r = await cliente.get(LAB + "/api/descargas")
        r.raise_for_status()
        lista = r.json()["archivos"]
        # El ritmo sale de las cabeceras del propio sitio: N pedidos cada V segundos (+10 % de margen).
        n = int(r.headers.get("X-RateLimit-Limit", "20"))
        v = int(r.headers.get("X-RateLimit-Window", "10"))
        ritmo = Ritmo(v / n * 1.1)
        for p in CARPETA.glob("*.part"):
            p.unlink()                            # restos de un corte anterior
        pendientes = [a for a in lista if not ((CARPETA / a["nombre"]).exists() and sha256(CARPETA / a["nombre"]) == a["sha256"])]
        saltados = len(lista) - len(pendientes)
        print(f"{len(lista)} archivos en el índice · {saltados} ya estaban (saltados) · {len(pendientes)} en cola")
        print(f"{TRABAJADORES} trabajadores · un pedido cada {ritmo.intervalo:.2f} s (límite del sitio: {n} cada {v} s)")
        if not pendientes:
            print("Nada que bajar: todo está y el sha256 cuadra.")
            return
        cola = asyncio.Queue()
        for a in pendientes:
            a["url"] = str(httpx.URL(LAB).join(a["url"]))  # la API da rutas absolutas del sitio
            cola.put_nowait(a)
        barra, fallidos = Barra(len(pendientes)), []
        tareas = [asyncio.create_task(trabajador(cola, cliente, ritmo, barra, fallidos)) for _ in range(TRABAJADORES)]
        await cola.join()
        for t in tareas:
            t.cancel()
        sys.stderr.write("\n")
        Path("fallidos.txt").write_text("\n".join(sorted(fallidos)) + ("\n" if fallidos else ""))
        print(f"Listo: {barra.ok} bajados, {saltados} saltados, {len(fallidos)} fallidos (en fallidos.txt) en {time.monotonic() - barra.t0:.0f} s")


if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print("\nCortado. Vuelve a correrlo: saltará lo que ya bajaste.")

"""04 · Límites: choca con el 429 a propósito, y luego aprende a no chocar.

Parte A · «a lo bruto»: pide sin pausa hasta recibir el primer 429 y muestra lo que dice el sitio
          (Retry-After, X-RateLimit-*). Se detiene en el primer 429, como haría un bot educado.
Parte B · «con modales»: espera lo que pidió el sitio y hace 30 pedidos leyendo X-RateLimit-Remaining
          y X-RateLimit-Reset para no volver a pasarse. Meta: cero 429.

    ALUMNO=tu-apodo CONTACTO=tu-correo python3 04_limites.py
    python3 04_limites.py --provocar-bloqueo   # ignora el 429 adrede para ver el 403 y su explicación (te bloquea ~1 min)
"""
import argparse
import os
import time
from collections import Counter

import httpx

LAB = os.environ.get("LAB", "https://jmd-learn.online/scraping-lab").rstrip("/")
ALUMNO = os.environ.get("ALUMNO", "sin-apodo")
CONTACTO = os.environ.get("CONTACTO", "sin-contacto")
UA = f"KitJMD-limites/1.0 ({ALUMNO}; {CONTACTO})"
URL = LAB + "/tienda?page=1"

ap = argparse.ArgumentParser()
ap.add_argument("--provocar-bloqueo", action="store_true")
ap.add_argument("--pedidos", type=int, default=30, help="pedidos de la parte B")
args = ap.parse_args()

c = httpx.Client(headers={"User-Agent": UA, "X-Alumno": ALUMNO, "Accept": "application/json"}, timeout=20)


def cab(r):
    h = r.headers
    return f"Remaining={h.get('X-RateLimit-Remaining', '?')} Reset={h.get('X-RateLimit-Reset', '-')} Retry-After={h.get('Retry-After', '-')}"

    if r.status_code != 200 or i % 5 == 0:

    # Si se me acaba el cupo, espero a que se renueve; si no, reparto lo que queda en el tiempo que falta.
    pausa = reinicio if quedan <= 1 else reinicio / quedan

"""05 · Tu scraper como herramienta MCP.

Un servidor MCP (FastMCP, transporte stdio) que expone la Tienda JMD de práctica a tu asistente
(Claude Code, OpenCode, Claude Desktop…). Por dentro usa la API con sesión del script 02 y respeta
Retry-After: el asistente pregunta en lenguaje natural y el scraper hace el trabajo con modales.

Herramientas:
  buscar_productos(texto, maximo)  → productos cuyo nombre, categoría o marca contiene el texto
  detalle(id)                      → ficha completa de un producto
  mi_estado()                      → tus pedidos, 429 y bloqueos según el laboratorio

Probar a mano (sin asistente):   python3 05_mcp_server.py --probar taza
Registrar en Claude Code:        claude mcp add tienda-jmd -e ALUMNO=tu-apodo -e CONTACTO=tu-correo -- python3 "$PWD/05_mcp_server.py"
(OpenCode y Claude Desktop: ver README.md)

IMPORTANTE con stdio: nunca uses print() a stdout, rompe el protocolo. Los mensajes van a stderr.
"""
import logging
import os
import re
import sys
import time

import httpx

LAB = os.environ.get("LAB", "https://jmd-learn.online/scraping-lab").rstrip("/")
ALUMNO = os.environ.get("ALUMNO", "sin-apodo")
CONTACTO = os.environ.get("CONTACTO", "sin-contacto")
UA = f"KitJMD-mcp/1.0 ({ALUMNO}; {CONTACTO})"

_cliente = None
logging.getLogger("httpx").setLevel(logging.WARNING)  # sin ruido en stderr


def log(*a):
    print(*a, file=sys.stderr, flush=True)


def _pedir(c, metodo, url, **kw):
    for intento in range(1, 5):
        r = c.request(metodo, url, **kw)
        if r.status_code in (429, 503):
            espera = min(int(r.headers.get("Retry-After", 2 ** intento)), 30)
            log(f"{r.status_code}: espero {espera} s")
            time.sleep(espera)
            continue
        return r
    raise RuntimeError("el laboratorio no responde, intenta en un minuto")


def _sesion():
    """Inicia sesión una sola vez (CSRF + cookie + token) y reutiliza el cliente."""
    global _cliente
    if _cliente is not None:
        return _cliente
    c = httpx.Client(base_url=LAB, headers={"User-Agent": UA, "X-Alumno": ALUMNO}, timeout=20, follow_redirects=True)
    html = _pedir(c, "GET", "/login").text
    csrf = re.search(r'name="csrf_token" value="([^"]+)"', html).group(1)
    r = _pedir(c, "POST", "/login", data={"csrf_token": csrf, "usuario": ALUMNO, "clave": "practica"})
    r.raise_for_status()
    c.headers["X-Lab-Token"] = re.search(r'id="api-token">([^<]+)<', r.text).group(1)
    log(f"sesión iniciada como {ALUMNO}")
    _cliente = c
    return c


def _api(ruta, **params):
    global _cliente
    for _ in range(2):
        r = _pedir(_sesion(), "GET", ruta, params=params)
        if r.status_code == 401:          # sesión vencida: entra de nuevo
            _cliente = None
            continue
        if r.status_code == 403:
            raise RuntimeError("bloqueado por el laboratorio: " + r.json().get("motivo", ""))
        r.raise_for_status()
        return r.json()
    raise RuntimeError("no se pudo iniciar sesión")


def buscar_productos(texto: str = "", maximo: int = 10) -> list[dict]:
    """Busca productos de la Tienda JMD de práctica (datos ficticios).

    texto: palabras a buscar (inicio de palabra, sin importar tildes) en nombre, categoría, marca o descripción. Vacío = todos.
    maximo: cuántos devolver como máximo (1 a 120).
    """
    maximo = max(1, min(int(maximo), 120))
    salida, cursor = [], None
    while len(salida) < maximo:
        params = {"q": texto, "limite": min(50, maximo - len(salida))}
        if cursor:
            params["cursor"] = cursor
        datos = _api("/api/productos", **params)
        salida += [{k: p[k] for k in ("id", "nombre", "categoria", "precio", "stock", "valoracion")} for p in datos["items"]]
        cursor = datos["next_cursor"]
        if not cursor:
            break
        time.sleep(0.5)                   # con modales también dentro del asistente
    return salida


def detalle(id: int) -> dict:
    """Ficha completa de un producto de la tienda de práctica: sku, marca, precio, stock, valoración, descripción."""
    return _api(f"/api/productos/{int(id)}")


def mi_estado() -> dict:
    """Cuántos pedidos, respuestas 429 y bloqueos lleva tu apodo en el laboratorio."""
    r = httpx.get(LAB + "/api/yo", headers={"User-Agent": UA, "X-Alumno": ALUMNO}, timeout=20)
    a = r.json()["alumno"]
    return {k: a.get(k) for k in ("nick", "pedidos", "c429", "c403", "bloqueos", "items_api", "archivos", "estado")}


if __name__ == "__main__":
    if len(sys.argv) > 1 and sys.argv[1] == "--probar":
        texto = sys.argv[2] if len(sys.argv) > 2 else ""
        res = buscar_productos(texto, 5)
        log(f"buscar_productos({texto!r}, 5) → {len(res)} resultados")
        for p in res:
            log(f"  #{p['id']} {p['nombre']} · S/ {p['precio']:.2f}")
        if res:
            d = detalle(res[0]["id"])
            log(f"detalle({res[0]['id']}) → {d['sku']} · {d['marca']} · {d['descripcion'][:60]}…")
        log(f"mi_estado() → {mi_estado()}")
        raise SystemExit(0)

    from mcp.server.fastmcp import FastMCP

    mcp = FastMCP("tienda-jmd")
    for f in (buscar_productos, detalle, mi_estado):
        mcp.tool()(f)
    mcp.run()                              # stdio por defecto
EOF

Las reglas del sitio
  • Máximo 20 pedidos cada 10 s por apodo. Si te pasas: 429 y Retry-After.
  • Si sigues tras más de 10 respuestas 429 en un minuto: 403 por 60 s.
  • /privado/ y /trampa/ están prohibidas en robots.txt. La trampa bloquea al instante.
  • Identifícate: apodo en X-Alumno, contacto en el User-Agent.

Ética y límites
  • Sí: leer robots.txt y términos, identificarte, ir despacio, obedecer Retry-After, preferir la API, guardar avance para no repetir pedidos.
  • No: rotar IPs o proxies, cambiar de apodo o de cuenta para esquivar un bloqueo, ni resolver CAPTCHAs. Si un sitio te frena, la respuesta es bajar el ritmo o pedir permiso.
  • El laboratorio te enseña cómo te detectan (conteo por apodo, trampas ocultas, ráfagas) para que entiendas por qué ir lento.
