Tu API con FastAPI

  1. Una carpeta nueva para tu API
    PS> mkdir mi-api; cd mi-api
  2. Crea el venv y actívalo · si PowerShell lo bloquea: Set-ExecutionPolicy -Scope CurrentUser RemoteSigned
    PS> py -m venv .venv
    PS> .venv\Scripts\Activate.ps1
  3. Instala FastAPI · con las comillas: los corchetes son parte del nombre
    (.venv) PS> pip install "fastapi[standard]"
  4. Crea main.py y pega el código de la derecha
    (.venv) PS> code main.py
  5. Levántala · se recarga sola cada vez que guardas
    (.venv) PS> fastapi dev main.py

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field

app = FastAPI(title="Inventario de mi tienda")


class Producto(BaseModel):
    """Lo que el cliente manda en el body de POST y PUT."""
    nombre: str
    precio: float = Field(gt=0)  # mayor que 0
    stock: int = Field(ge=0)     # 0 o más
    categoria: str


# La «base de datos»: un dict en memoria. Llave = id, valor = el producto
productos: dict[int, dict] = {}
siguiente_id = 1


def buscar(producto_id: int) -> dict:
    if producto_id not in productos:  # en un dict se va directo a la llave
        raise HTTPException(status_code=404, detail="Producto no encontrado")
    return productos[producto_id]


@app.get("/productos")
def listar():
    return list(productos.values())


@app.get("/productos/{producto_id}")
def ver(producto_id: int):
    return buscar(producto_id)


@app.post("/productos", status_code=201)
def crear(producto: Producto):
    global siguiente_id
    nuevo = {"id": siguiente_id, **producto.model_dump()}  # id + los 4 campos
    productos[siguiente_id] = nuevo
    siguiente_id += 1
    return nuevo


@app.put("/productos/{producto_id}")
def editar(producto_id: int, producto: Producto):
    buscar(producto_id)  # si no existe, responde 404
    productos[producto_id] = {"id": producto_id, **producto.model_dump()}
    return productos[producto_id]


@app.delete("/productos/{producto_id}", status_code=204)
def borrar(producto_id: int):
    buscar(producto_id)
    del productos[producto_id]
