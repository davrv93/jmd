s.alNota(n => {
  const p = s.aPantalla(n.boca);   // boca → píxeles
  const e = document.createElement('span');
  e.className = 'nota'; e.textContent = '♪';
  e.style.left = p.x + 'px'; e.style.top = p.y + 'px';
  escena.appendChild(e);          // y CSS la anima
});
