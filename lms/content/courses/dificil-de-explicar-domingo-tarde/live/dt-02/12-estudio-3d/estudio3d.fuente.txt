function geoCiclorama(ancho = 40, frente = 14, fondoZ = -7.5, R = 3.2, alto = 16) {
  const pts = [];
  const zc = fondoZ + R;
  pts.push([frente, 0]);
  for (let i = 0; i <= 24; i++) { const a = (i / 24) * Math.PI / 2; pts.push([zc - Math.sin(a) * R, R - Math.cos(a) * R]); }
  pts.push([fondoZ, alto]);
  const pos = [], uv = [], idx = [];
  pts.forEach(([z, y], i) => { pos.push(-ancho / 2, y, z, ancho / 2, y, z); uv.push(0, i / (pts.length - 1), 1, i / (pts.length - 1)); });
  for (let i = 0; i < pts.length - 1; i++) { const a = i * 2; idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2); }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  g.setIndex(idx); g.computeVertexNormals();
  return g;
}

// ---------- Entorno: HDRI (RGBE en WebP sin pérdida) + softboxes ----------
const cacheHDR = new Map();
async function cargarHDR(nombre) {
  if (cacheHDR.has(nombre)) return cacheHDR.get(nombre);
  const p = (async () => {
    const r = await fetch(`./hdri-${nombre}.webp`);
    if (!r.ok) throw new Error('HDRI ' + r.status);
    const bmp = await createImageBitmap(await r.blob(), { colorSpaceConversion: 'none', premultiplyAlpha: 'none' });
    const W = bmp.width, H = bmp.height / 2;
    const c = document.createElement('canvas'); c.width = bmp.width; c.height = bmp.height;
    const g = c.getContext('2d', { willReadFrequently: true });
    g.drawImage(bmp, 0, 0);
    const px = g.getImageData(0, 0, bmp.width, bmp.height).data;
    const out = new Uint16Array(W * H * 4), toH = THREE.DataUtils.toHalfFloat;
    const off = W * H * 4;
    for (let i = 0, n = W * H; i < n; i++) {
      const e = px[off + i * 4];
      const k = e ? Math.pow(2, e - 136) : 0;
      // Tope de 24: las lámparas desnudas del HDRI (hasta 3000) quemarían la superficie pulida.
      out[i * 4] = toH(Math.min(24, (px[i * 4] + 0.5) * k)); out[i * 4 + 1] = toH(Math.min(24, (px[i * 4 + 1] + 0.5) * k)); out[i * 4 + 2] = toH(Math.min(24, (px[i * 4 + 2] + 0.5) * k)); out[i * 4 + 3] = 15360;
    }
    const t = new THREE.DataTexture(out, W, H, THREE.RGBAFormat, THREE.HalfFloatType);
    t.mapping = THREE.EquirectangularReflectionMapping; t.flipY = false;
    t.magFilter = THREE.LinearFilter; t.minFilter = THREE.LinearFilter; t.generateMipmaps = false;
    t.colorSpace = THREE.LinearSRGBColorSpace; t.needsUpdate = true;
    return t;
  })();
  cacheHDR.set(nombre, p);
  return p;
}

async function construirEntorno() {
  const P = PRESETS[est.luz];
  const clave_ = est.luz + (est.calido ? 'c' : '');
  let hdr = null;
  try { hdr = await cargarHDR(P.hdri); } catch (e) { console.warn('HDRI no disponible, uso softboxes', e.message); }
  if (clave_ === envClave && envRT) return;
  esferaEnv.material.map = hdr; esferaEnv.material.color.setScalar(hdr ? P.envI : 0.02); esferaEnv.material.needsUpdate = true;
  // El equirectangular de PMREM espera la costura detrás: giramos para que la ventana quede a un lado.
  esferaEnv.rotation.y = P.hdri === 'ventana' ? -Math.PI / 2 : 0;
  formasEnv.clear();
  for (const [pos, [w, h], int, col] of P.forms) {
    const m = new THREE.Mesh(new THREE.PlaneGeometry(w, h), new THREE.MeshBasicMaterial({ color: col || '#ffffff', toneMapped: false, side: THREE.DoubleSide }));
    m.material.color.multiplyScalar(int * (est.calido && !col ? 1 : 1));
    if (est.calido && !col) m.material.color.multiply(new THREE.Color('#ffd9ae'));
    m.position.set(...pos).multiplyScalar(1.6); m.lookAt(0, 1, 0);
    formasEnv.add(m);
  }
  const nuevo = pmrem.fromScene(envScene, 0, 0.1, 100);
  if (envRT) envRT.dispose();
  envRT = nuevo; envClave = clave_;
  scene.environment = envRT.texture;
  invalidar();
}

function renderContacto() {
  if (!actual) return;
  const caja = new THREE.Box3().setFromObject(actual.g), c = caja.getCenter(new THREE.Vector3()), s = caja.getSize(new THREE.Vector3());
  const lado = Math.max(s.x, s.z) * 1.7 + 0.6;
  camCS.left = -lado / 2; camCS.right = lado / 2; camCS.top = lado / 2; camCS.bottom = -lado / 2;
  camCS.far = Math.max(0.6, Math.min(1.6, s.y * 0.6)); camCS.updateProjectionMatrix();
  camCS.position.set(c.x, SUP_Y, c.z);
  contacto.position.set(c.x, SUP_Y + 0.003, c.z); contacto.scale.set(lado, -lado, 1);
  const vis = [];
  scene.traverse((o) => { if ((o.isMesh || o.isLine) && o.visible && !esDelProducto(o)) { vis.push(o); o.visible = false; } });
  const bgPrev = scene.background, ovPrev = scene.overrideMaterial, ccPrev = renderer.getClearAlpha();
  scene.background = null; scene.overrideMaterial = matProf;
  renderer.setClearColor(0x000000, 0);
  renderer.setRenderTarget(rtCS); renderer.clear(); renderer.render(scene, camCS);
  scene.overrideMaterial = ovPrev; scene.background = bgPrev;
  vis.forEach((o) => { o.visible = true; });
  const pasos = [1.2, 2.4];
  for (const b of pasos) {
    quadBlur.material = blurH; blurH.uniforms.tDiffuse.value = rtCS.texture; blurH.uniforms.h.value = b / 256;
    renderer.setRenderTarget(rtCSb); quadBlur.render(renderer);
    quadBlur.material = blurV; blurV.uniforms.tDiffuse.value = rtCSb.texture; blurV.uniforms.v.value = b / 256;
    renderer.setRenderTarget(rtCS); quadBlur.render(renderer);
  }
  renderer.setRenderTarget(null); renderer.setClearColor(0x000000, ccPrev);
  csSucio = false;
}

function perfilSuave(puntos, pasos = 6) {
  // Redondea las esquinas de un perfil (x = radio, y = altura) con curvas de Catmull-Rom.
  const curva = new THREE.SplineCurve(puntos.map(([x, y]) => new THREE.Vector2(x, y)));
  return curva.getPoints(puntos.length * pasos);
}
function torno(puntos, seg = 96, suave = true) {
  const p = suave ? perfilSuave(puntos) : puntos.map(([x, y]) => new THREE.Vector2(x, y));
  return new THREE.LatheGeometry(p, seg);
}

  });
}

// ---------- Cámara ----------
let viaje = null;
function encuadre() {
  const caja = new THREE.Box3().setFromObject(actual.g), s = caja.getSize(new THREE.Vector3()), c = caja.getCenter(new THREE.Vector3());
  const vfov = THREE.MathUtils.degToRad(camera.fov), asp = Math.max(0.6, camera.aspect);
  const altoVis = Math.max(s.y * 1.7, (Math.max(s.x, s.z) * 1.35) / asp);
  const d = altoVis / 2 / Math.tan(vfov / 2);
  return { d, c, s };
}
function posCamara(tipo) {
  const { d, c, s } = encuadre();
  const objetivo = new THREE.Vector3(c.x, c.y + s.y * 0.06, c.z);
  if (tipo === 'frente') return [new THREE.Vector3(c.x, objetivo.y + d * 0.08, c.z + d), objetivo];
  if (tipo === 'cenital') { const dd = Math.max(s.x, s.z) * 1.6 / 2 / Math.tan(THREE.MathUtils.degToRad(camera.fov) / 2) / Math.min(1, camera.aspect); return [new THREE.Vector3(c.x, dd, c.z + 0.02), new THREE.Vector3(c.x, 0, c.z)]; }
  const az = 0.42, el = actual.el ?? 0.2;
  return [new THREE.Vector3(c.x + Math.sin(az) * Math.cos(el) * d, objetivo.y + Math.sin(el) * d, c.z + Math.cos(az) * Math.cos(el) * d), objetivo];
}
function irCamara(tipo, inmediato) {
  const [p, t] = posCamara(tipo);
  controls.maxPolarAngle = tipo === 'cenital' ? Math.PI / 2 : Math.PI / 2 - 0.04;
  // Regla de tercios: el producto se corre a la izquierda dejando aire hacia la luz.
  camera.filmOffset = tipo === 'cenital' ? 0 : 0.09 * camera.getFilmWidth() * 2 * Math.tan(THREE.MathUtils.degToRad(camera.fov / 2)) * camera.aspect / Math.max(1, camera.aspect);
  camera.updateProjectionMatrix();
  if (inmediato) { camera.position.copy(p); controls.target.copy(t); controls.update(); viaje = null; invalidar(); return; }
  viaje = { p0: camera.position.clone(), t0: controls.target.clone(), p1: p, t1: t, ini: performance.now(), dur: 650 };
}

// ---------- Calidad adaptable ----------
const NIVELES = { 3: 'alta', 2: 'media', 1: 'ligera' };
function aplicarNivel(n) {
  est.nivel = n;
  pBloom.enabled = n >= 3;
  pBokeh.enabled = est.desenfoque && n >= 2;
  pGrano.enabled = n >= 2;
  clave.castShadow = n >= 2;
  const sm = n >= 3 ? 2048 : 1024;
  if (clave.shadow.mapSize.x !== sm) { clave.shadow.mapSize.set(sm, sm); if (clave.shadow.map) { clave.shadow.map.dispose(); clave.shadow.map = null; } }
  ajustarTam();
  $('nivel').textContent = 'Calidad ' + NIVELES[n] + (est.calidad === 'auto' ? ' · automática' : '');
}
const muestras = [];
let ultimo = 0, pausaMedida = 0;
function medir(ahora) {
  if (est.calidad !== 'auto') return;
  if (ultimo && ahora - ultimo < 250) muestras.push(ahora - ultimo);
  ultimo = ahora;
  if (muestras.length >= 30 && ahora > pausaMedida) {
    const prom = muestras.reduce((a, b) => a + b, 0) / muestras.length;
    muestras.length = 0;
    if (prom > 36 && est.nivel > 1) { aplicarNivel(est.nivel - 1); pausaMedida = ahora + 1500; }
  }
}

let ancho = 0, alto = 0;
function ajustarTam() {
  const r = vista.getBoundingClientRect();
  ancho = Math.max(200, Math.floor(r.width)); alto = Math.max(160, Math.floor(r.height));
  const pr = est.nivel >= 3 ? Math.min(window.devicePixelRatio || 1, 2) : est.nivel === 2 ? Math.min(window.devicePixelRatio || 1, 1.25) : 1;
  renderer.setPixelRatio(pr); renderer.setSize(ancho, alto, false);
  renderer.domElement.style.width = '100%'; renderer.domElement.style.height = '100%';
  composer.setPixelRatio(pr); composer.setSize(ancho, alto);
  camera.aspect = ancho / alto; camera.updateProjectionMatrix();
  if (actual && !viaje) irCamaraSuave();
  invalidar();
}
let tAjuste = 0;
function irCamaraSuave() { clearTimeout(tAjuste); tAjuste = setTimeout(() => { camera.filmOffset = est.camara === 'cenital' ? 0 : 0.09 * camera.getFilmWidth() * 2 * Math.tan(THREE.MathUtils.degToRad(camera.fov / 2)) * camera.aspect / Math.max(1, camera.aspect); camera.updateProjectionMatrix(); invalidar(); }, 50); }

let pendiente = true;
function invalidar() { pendiente = true; }
controls.addEventListener('change', invalidar);
function dibujar() {
  if (csSucio) renderContacto();
  pBokeh.uniforms.focus.value = camera.position.distanceTo(controls.target);
  pBokeh.uniforms.aperture.value = 0.00055 * (9 / Math.max(3, pBokeh.uniforms.focus.value));
  pGrano.uniforms.uSemilla.value = (pGrano.uniforms.uSemilla.value + 1.37) % 100;
  composer.render();
}
function bucle(ahora) {
  requestAnimationFrame(bucle);
  let mueve = false;
  if (viaje) {
    const t = Math.min(1, (ahora - viaje.ini) / viaje.dur), e = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
    camera.position.lerpVectors(viaje.p0, viaje.p1, e); controls.target.lerpVectors(viaje.t0, viaje.t1, e);
    if (t >= 1) viaje = null;
    mueve = true;
  }
  if (est.giro && actual) { actual.g.rotation.y += 0.005; csSucio = true; mueve = true; }
  const amort = controls.update();
  if (mueve || amort || pendiente) {
    pendiente = false;
    dibujar();
    if (mueve || amort) medir(ahora); else ultimo = 0;
  }
}

// ---------- Exportar PNG ----------
async function exportar(escala, transparente) {
  const max = renderer.capabilities.maxTextureSize || 4096;
  let w = Math.round(ancho * escala), h = Math.round(alto * escala);
  const tope = Math.min(4096, max), k = Math.min(1, tope / Math.max(w, h));
  w = Math.round(w * k); h = Math.round(h * k);
  const prPrev = renderer.getPixelRatio();
  renderer.setPixelRatio(1); renderer.setSize(w, h, false);
  let blob;
  const ocultos = transparente ? [ciclorama, superficie] : [];
  try {
    ocultos.forEach((o) => { o.visible = false; });
    if (transparente) renderer.setClearColor(0x000000, 0);
    composer.setPixelRatio(1); composer.setSize(w, h);
    const bok = pBokeh.enabled, gr = pGrano.enabled, bl = pBloom.enabled;
    if (transparente) { pBokeh.enabled = false; pGrano.enabled = false; pBloom.enabled = false; }
    if (transparente) { renderer.render(scene, camera); } else { pBokeh.enabled = est.desenfoque; pGrano.enabled = true; pBloom.enabled = true; dibujar(); }
    pBokeh.enabled = bok; pGrano.enabled = gr; pBloom.enabled = bl;
    blob = await new Promise((ok) => renderer.domElement.toBlob(ok, 'image/png'));
  } finally {
    ocultos.forEach((o) => { o.visible = true; });
    renderer.setPixelRatio(prPrev);
    ajustarTam();
    aplicarNivel(est.nivel);
    dibujar();
  }
  return { blob, w, h };
}

function clasificar(texto) {
  const t = norm(texto || '');
  if (window.clasificarRubroEstilo) { try { const r = window.clasificarRubroEstilo(texto); if (r && r.rubro) return r; } catch (e) { /* usamos el propio */ } }
  const rubro = (RUBROS.find(([, re]) => re.test(t)) || [])[0] || 'empresa';
  let estilo = (ESTILOS.find(([, re]) => re.test(t)) || [])[0];
  const colores = COLORES.filter(([re]) => re.test(t));
  return { rubro, estilo, colores };
}
function describir(texto) {
  const { rubro, estilo: est0, colores } = clasificar(texto);
  const [plantilla, estiloDef] = RUBRO_A[rubro] || RUBRO_A.empresa;
  const estilo = est0 || (rubro === 'postres' ? null : estiloDef);
  const P = PLANTILLAS[plantilla];
  const op = { plantilla, color: P.color, material: P.material, luz: P.luz, sup: P.sup, acento: P.acento, fondo: PRESETS[P.luz].fondo, calido: !!P.calido };
  if (estilo && MAPA_ESTILO[estilo]) Object.assign(op, { calido: false }, MAPA_ESTILO[estilo], estilo === 'lujo' ? { fondo: PRESETS.lujo.fondo } : {});
  if (rubro === 'postres' && !est0) Object.assign(op, { luz: 'gourmet', sup: 'madera', fondo: PRESETS.gourmet.fondo });
  // Paleta: el primer color no metálico pinta el producto; un metal mencionado pasa a acento.
  const metales = colores.filter((c) => c[2] && c[2] !== 'negro');
  const tonos = colores.filter((c) => !c[2] || c[2] === 'negro');
  if (metales[0]) op.acento = metales[0][2];
  if (tonos[0]) {
    let col = tonos[0][1];
    if (plantilla === 'perfume' && tonos[0][2] === 'negro') { col = '#4a2a0c'; op.sup = 'marmol-negro'; }
    if (plantilla !== 'plato') op.color = col;
  } else if (metales[0] && plantilla === 'perfume') op.color = '#a8641c';
  if (plantilla === 'logo' && metales[0]) { op.material = 'metal'; op.color = metales[0][1]; }
  return { op, rubro, estilo: estilo || (rubro === 'postres' ? 'gourmet' : estiloDef) };
}

// ---------- Controles de la página ----------
function marcar(grupo, valor) {
  document.querySelectorAll(`[data-g="${grupo}"] button`).forEach((b) => b.setAttribute('aria-pressed', String(b.dataset.v === valor)));
}
function sincronizar() {
  ['plantilla', 'material', 'luz', 'sup', 'acento', 'camara'].forEach((g) => marcar(g, est[g]));
  $('color').value = est.color; $('colorFondo').value = est.fondo;
  const P = PLANTILLAS[est.plantilla];
  $('marca').placeholder = P.marca; $('linea').placeholder = P.linea || 'Sin línea secundaria';
  $('linea').disabled = est.plantilla === 'logo';
}
function aplicar(op) {
  const cambiaPlantilla = op.plantilla && op.plantilla !== est.plantilla;
  if (op.plantilla) {
    const P = PLANTILLAS[op.plantilla];
    const base = { color: P.color, material: P.material, luz: P.luz, sup: P.sup, acento: P.acento, fondo: P.fondo, calido: !!P.calido };
    Object.assign(est, cambiaPlantilla ? base : {}, op);
  } else Object.assign(est, op);
  if (op.calidad) aplicarNivel(op.calidad === 'auto' ? 3 : +op.calidad);
  if ('desenfoque' in op) { $('desenfoque').checked = est.desenfoque; pBokeh.enabled = est.desenfoque && est.nivel >= 2; }
  aplicarSuperficie(); aplicarLuz(); reconstruir(); irCamara(est.camara, true);
  sincronizar();
}
function elegirPlantilla(p) { aplicar({ plantilla: p, marca: '', linea: '' }); $('marca').value = ''; $('linea').value = ''; }
document.querySelectorAll('[data-g] button').forEach((b) => {
  b.type = 'button';
  b.addEventListener('click', () => {
    const g = b.parentElement.dataset.g, v = b.dataset.v;
    if (g === 'plantilla') return elegirPlantilla(v);
    if (g === 'escala') return marcar('escala', v);
    if (g === 'calidad') { est.calidad = v; marcar('calidad', v); aplicarNivel(v === 'auto' ? 3 : +v); invalidar(); return; }
    est[g] = v; marcar(g, v);
    if (g === 'material' || g === 'acento') reconstruir();
    else if (g === 'sup') aplicarSuperficie();
    else if (g === 'luz') { est.fondo = PRESETS[v].fondo; est.calido = false; $('colorFondo').value = est.fondo; aplicarLuz(); aplicarSuperficie(); }
    else if (g === 'camara') irCamara(v);
  });
});
let tTexto = 0;
$('color').addEventListener('input', (e) => { est.color = e.target.value; clearTimeout(tTexto); tTexto = setTimeout(reconstruir, 60); });
$('colorFondo').addEventListener('input', (e) => { est.fondo = e.target.value; aplicarFondo(); });
for (const id of ['marca', 'linea']) $(id).addEventListener('input', (e) => { est[id] = e.target.value; clearTimeout(tTexto); tTexto = setTimeout(reconstruir, 250); });
$('desenfoque').addEventListener('change', (e) => { est.desenfoque = e.target.checked; pBokeh.enabled = est.desenfoque && est.nivel >= 2; invalidar(); });
$('giro').addEventListener('change', (e) => { est.giro = e.target.checked; if (!est.giro && actual) { actual.g.rotation.y = 0; csSucio = true; invalidar(); } });
function usarDescripcion() {
  const txt = $('describe').value.trim();
  if (!txt) { avisar('Escribe qué vendes y cómo lo imaginas, por ejemplo «perfume árabe de lujo, dorado y negro».'); return; }
  const r = describir(txt);
  aplicar(r.op);
  const NOM = { lujo: 'Lujo oscuro', ventana: 'Luz de ventana', blanco: 'Estudio blanco', gourmet: 'Cálido gourmet', tecno: 'Tecnológico frío' };
  $('lectura').textContent = `Rubro: ${r.rubro} · estilo: ${r.estilo} → ${NOM[est.luz]}${est.calido ? ' cálida' : ''}, ${est.sup.replace('-', ' ')}, acento ${est.acento}.`;
}
$('aplicarDesc').addEventListener('click', usarDescripcion);
$('describe').addEventListener('keydown', (e) => { if (e.key === 'Enter') { e.preventDefault(); usarDescripcion(); } });
$('imagen').addEventListener('change', (e) => {
  const f = e.target.files && e.target.files[0];
  if (!f) return;
  if (f.size > 8 * 1024 * 1024) { avisar('La imagen pesa más de 8 MB: usa una más liviana.'); return; }
  const lector = new FileReader();
  lector.onload = () => {
    const im = new Image();
    im.onload = () => { est.imagen = im; $('quitar').hidden = false; reconstruir(); avisar('Etiqueta cambiada por tu imagen.'); };
    im.onerror = () => avisar('No pude leer esa imagen: prueba con PNG, JPG o WebP.');
    im.src = lector.result;
  };
  lector.readAsDataURL(f);
});
$('quitar').addEventListener('click', () => { est.imagen = null; $('imagen').value = ''; $('quitar').hidden = true; reconstruir(); });
function avisar(t) { const a = $('aviso'); a.textContent = t; clearTimeout(avisar.t); avisar.t = setTimeout(() => { a.textContent = ''; }, 5000); }
$('descargar').addEventListener('click', async () => {
  const b = $('descargar'); b.disabled = true; const txt = b.textContent; b.textContent = 'Renderizando…';
  try {
    const escala = +document.querySelector('[data-g="escala"] [aria-pressed="true"]').dataset.v;
    const { blob, w, h } = await exportar(escala, $('transparente').checked);
    if (!blob) throw new Error('sin imagen');
    const url = URL.createObjectURL(blob), a = document.createElement('a');
    a.href = url; a.download = `estudio3d-${est.plantilla}-${w}x${h}.png`; document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 4000);
    avisar(`Descargado: ${w} × ${h} px.`);
  } catch (err) {
    avisar('No se pudo exportar a ese tamaño: prueba con 2x.');
  } finally { b.disabled = false; b.textContent = txt; }
});

// API para pruebas y para el video de la clase
window.estudio3d = {
  async set(op) {
    aplicar(op);
    await fuentesListas;
    try { await cargarHDR(PRESETS[est.luz].hdri); } catch (e) { /* sigue con softboxes */ }
    await construirEntorno();
    await new Promise((r) => setTimeout(r, 30));
    reconstruir(); irCamara(est.camara, true);
    dibujar();
    return { nivel: est.nivel, w: ancho, h: alto };
  },
  describir(texto) { $('describe').value = texto; usarDescripcion(); return describir(texto); },
  png() { dibujar(); return renderer.domElement.toDataURL('image/png'); },
  async pngExport(escala) { const r = await exportar(escala, false); return await new Promise((ok) => { const f = new FileReader(); f.onload = () => ok(f.result); f.readAsDataURL(r.blob); }); },
  async exportar(escala, transparente) { const r = await exportar(escala, transparente); return { w: r.w, h: r.h, bytes: r.blob ? r.blob.size : 0 }; },
  estado: () => ({ ...est, imagen: !!est.imagen }),
  _interno: { clave, relleno, rims, luzFondo, supMat, cicloMat, contacto, scene, invalidar: () => invalidar() },
};

// Prueba de rendimiento al abrir: 20 cuadros; si el equipo es lento, baja sola la calidad.
(function prueba() {
  let n = 0, t0 = 0;
  const paso = (t) => {
    if (est.calidad !== 'auto') return;
    if (n === 4) t0 = t;
    if (n < 24) { n++; dibujar(); requestAnimationFrame(paso); return; }
    const prom = (t - t0) / 20;
    if (prom > 55) aplicarNivel(1); else if (prom > 30) aplicarNivel(2);
    invalidar();
  };
  setTimeout(() => requestAnimationFrame(paso), 400);
})();
