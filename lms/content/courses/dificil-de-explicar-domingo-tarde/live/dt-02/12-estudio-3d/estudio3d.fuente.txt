  const t = new THREE.CanvasTexture(c); t.colorSpace = THREE.SRGBColorSpace; return t;
}

// ---------- Utilidades de color y texturas ----------
function lum(hex) { const c = new THREE.Color(hex); return 0.2126 * c.r + 0.7152 * c.g + 0.0722 * c.b; }
function mezcla(hex, otro, t) { return '#' + new THREE.Color(hex).lerp(new THREE.Color(otro), t).getHexString(); }

function texFondo() {
  const c = document.createElement('canvas'); c.width = 64; c.height = 512;
  const g = c.getContext('2d');
  let a, b;
  if (est.fondo === 'estudio') { a = '#f4f2ee'; b = '#c9c4bb'; if (est.luz === 'dramatica') { a = '#4a4744'; b = '#151413'; } }
  else if (est.fondo === 'degradado') { a = mezcla(est.colorMarca, '#ffffff', 0.45); b = mezcla(est.colorMarca, '#000000', 0.45); }
  else { a = est.colorMarca; b = est.colorMarca; }
  const lin = g.createLinearGradient(0, 0, 0, 512);
  lin.addColorStop(0, a); lin.addColorStop(1, b);
  g.fillStyle = lin; g.fillRect(0, 0, 64, 512);
  const t = new THREE.CanvasTexture(c); t.colorSpace = THREE.SRGBColorSpace; return t;
}
let fondoTex = null;
function aplicarFondo() {
  if (fondoTex) fondoTex.dispose();
  fondoTex = texFondo();
  scene.background = fondoTex;

}

// Etiqueta: texto de marca o la imagen que suba el alumno (cubre la etiqueta entera).
function canvasEtiqueta(w, h, fondo, opts = {}) {
  const c = document.createElement('canvas'); c.width = w; c.height = h;
  const g = c.getContext('2d');
  if (est.imagen) {
    const im = est.imagen, s = Math.max(w / im.width, h / im.height);
    g.fillStyle = '#ffffff'; g.fillRect(0, 0, w, h);
    g.drawImage(im, (w - im.width * s) / 2, (h - im.height * s) / 2, im.width * s, im.height * s);
    return c;
  }
  g.fillStyle = fondo; g.fillRect(0, 0, w, h);
  const tinta = lum(fondo) > 0.45 ? '#1b1a17' : '#f7f3ea';
  const marca = (est.marca || PLANTILLAS[est.plantilla].marca).trim() || 'Tu marca';
  const linea = est.linea !== '' ? est.linea : PLANTILLAS[est.plantilla].linea;
  const cx = opts.cx ?? w / 2, ancho = opts.ancho ?? w * 0.82;
  g.textAlign = 'center'; g.textBaseline = 'middle'; g.fillStyle = tinta;
  let t = Math.round(h * (opts.escala ?? 0.26));
  const familia = opts.serif === false ? '"Helvetica Neue", Arial, sans-serif' : 'Georgia, "Times New Roman", serif';
  do { g.font = `600 ${t}px ${familia}`; t -= 4; } while (g.measureText(marca).width > ancho && t > 12);
  g.fillText(marca, cx, h * (linea ? 0.44 : 0.5));
  if (linea) {
    let s = Math.round(h * 0.075);
    do { g.font = `500 ${s}px "Helvetica Neue", Arial, sans-serif`; s -= 2; } while (g.measureText(linea.toUpperCase()).width > ancho && s > 8);
    g.globalAlpha = 0.78; g.fillText(linea.toUpperCase(), cx, h * 0.68); g.globalAlpha = 1;
    g.fillRect(cx - w * 0.06, h * 0.58, w * 0.12, Math.max(2, h * 0.006));
  }
  if (opts.borde) { g.strokeStyle = tinta; g.globalAlpha = 0.35; g.lineWidth = Math.max(2, w * 0.006); g.strokeRect(w * 0.05, h * 0.07, w * 0.9, h * 0.86); g.globalAlpha = 1; }
  return c;
}
function tex(c, aniso = 8) {
  const t = new THREE.CanvasTexture(c); t.colorSpace = THREE.SRGBColorSpace;
  t.anisotropy = Math.min(aniso, renderer.capabilities.getMaxAnisotropy()); return t;
}

// Materiales: un MeshPhysicalMaterial que cambia de «acabado» sin crear otro.
function aplicarAcabado(m, tipo, color) {
  const c = new THREE.Color(color);
  Object.assign(m, { metalness: 0, roughness: 0.5, transmission: 0, thickness: 0, ior: 1.5, clearcoat: 0, clearcoatRoughness: 0.1, sheen: 0, specularIntensity: 1, transparent: false, opacity: 1 });
  m.attenuationDistance = Infinity; m.attenuationColor.set('#ffffff');
  m.color.copy(c);
  if (tipo === 'vidrio') {
    Object.assign(m, { transmission: 1, roughness: 0.04, thickness: 0.9, ior: 1.5, clearcoat: 1, clearcoatRoughness: 0.03 });
    m.color.set('#ffffff'); m.attenuationColor.copy(c); m.attenuationDistance = 0.55;
  } else if (tipo === 'brillante') {
    Object.assign(m, { roughness: 0.3, clearcoat: 1, clearcoatRoughness: 0.04 });
  } else if (tipo === 'mate') {
    Object.assign(m, { roughness: 0.88 });
  } else if (tipo === 'metal') {
    Object.assign(m, { metalness: 1, roughness: 0.24, clearcoat: 0.4, clearcoatRoughness: 0.12 });
  } else if (tipo === 'satinado') {
    Object.assign(m, { roughness: 0.46, sheen: 1, clearcoat: 0.35, clearcoatRoughness: 0.3 });
    m.sheenColor.set('#ffffff'); m.sheenRoughness = 0.5;
  }
  m.needsUpdate = true;
}
const fis = (o) => new THREE.MeshPhysicalMaterial(o);
const metalOro = () => fis({ color: '#d4ae68', metalness: 1, roughness: 0.2, clearcoat: 0.5, clearcoatRoughness: 0.1 });
const metalPlata = () => fis({ color: '#d5d8dd', metalness: 1, roughness: 0.28 });
function malla(geo, mat, x = 0, y = 0, z = 0) {
  const m = new THREE.Mesh(geo, mat); m.position.set(x, y, z); m.castShadow = true; m.receiveShadow = true; return m;
}

let actual = null;
function liberar(obj) {
  obj.traverse((o) => {
    if (o.geometry) o.geometry.dispose();
    const ms = Array.isArray(o.material) ? o.material : o.material ? [o.material] : [];
    ms.forEach((m) => { if (m.map) m.map.dispose(); m.dispose(); });
  });
}
function reconstruir() {
  if (actual) { scene.remove(actual.g); liberar(actual.g); }
  actual = construir[est.plantilla]();
  scene.add(actual.g);
  contacto.scale.set(actual.huella * 1.15, actual.huella * 0.95, 1);
  pBokeh.uniforms.focus.value = camera.position.distanceTo(controls.target);
  invalidar();
}

// ---------- Luz y cámara ----------
function aplicarLuz() {
  const L = est.luz;
  clave.color.set(L === 'calida' ? '#ffd2a1' : '#ffffff');
  clave.intensity = L === 'dramatica' ? 4.2 : L === 'calida' ? 2.4 : 1.8;
  clave.position.set(L === 'dramatica' ? 5.5 : 3.5, L === 'dramatica' ? 4.2 : 6.5, L === 'dramatica' ? 1.5 : 4.5);
  contra.intensity = L === 'dramatica' ? 3.0 : L === 'calida' ? 1.0 : 1.1;
  contra.color.set(L === 'calida' ? '#ffb36b' : '#ffffff');
  relleno.intensity = L === 'calida' ? 9 : 0;
  scene.environmentIntensity = L === 'dramatica' ? 0.28 : L === 'calida' ? 0.75 : 1.0;
  suelo.material.opacity = L === 'dramatica' ? 0.42 : 0.26;
  renderer.toneMappingExposure = L === 'dramatica' ? 1.05 : 1.0;
  aplicarFondo();
  invalidar();
}
let viaje = null;
function posCamara(tipo) {
  const P = PLANTILLAS[est.plantilla], d = P.dist, f = actual ? actual.foco : 0.8;
  const objetivo = new THREE.Vector3(0, f, 0);
  if (tipo === 'frente') return [new THREE.Vector3(0, f + 0.35, d), objetivo];
  if (tipo === 'cenital') return [new THREE.Vector3(0, d * 1.05, 0.02), new THREE.Vector3(0, 0.3, 0)];
  return [new THREE.Vector3(d * 0.6, f + d * 0.3, d * 0.76), objetivo];
}
function irCamara(tipo, inmediato) {
  const [p, t] = posCamara(tipo);
  controls.maxPolarAngle = tipo === 'cenital' ? Math.PI / 2 : Math.PI / 2 - 0.03;
  if (inmediato) { camera.position.copy(p); controls.target.copy(t); controls.update(); viaje = null; invalidar(); return; }
  viaje = { p0: camera.position.clone(), t0: controls.target.clone(), p1: p, t1: t, ini: performance.now(), dur: 650 };
}

// ---------- Calidad adaptable ----------
const NIVELES = { 3: 'alta', 2: 'media', 1: 'ligera' };
function aplicarNivel(n) {
  est.nivel = n;
  pAO.enabled = n >= 3;
  pBloom.enabled = n >= 2;
  pBloom.strength = n >= 3 ? 0.16 : 0.12;
  pBokeh.enabled = est.desenfoque && n >= 2;
  const sm = n >= 2 ? 2048 : 1024;
  if (clave.shadow.mapSize.x !== sm) { clave.shadow.mapSize.set(sm, sm); if (clave.shadow.map) { clave.shadow.map.dispose(); clave.shadow.map = null; } }
  ajustarTam();
  $('nivel').textContent = 'Calidad ' + NIVELES[n] + (est.calidad === 'auto' ? ' · automática' : '');
}
const muestras = [];
let ultimo = 0, pausaMedida = 0;
function medir(ahora) {
  if (est.calidad !== 'auto') return;
  if (ultimo && ahora - ultimo < 200) muestras.push(ahora - ultimo);
  ultimo = ahora;
  if (muestras.length >= 30 && ahora > pausaMedida) {
    const prom = muestras.reduce((a, b) => a + b, 0) / muestras.length;
    muestras.length = 0;
    if (prom > 34 && est.nivel > 1) { aplicarNivel(est.nivel - 1); pausaMedida = ahora + 1500; }
  }
}

// ---------- Tamaño y bucle ----------
let ancho = 0, alto = 0;
function ajustarTam() {
  const r = vista.getBoundingClientRect();
  ancho = Math.max(200, Math.floor(r.width)); alto = Math.max(160, Math.floor(r.height));
  const pr = est.nivel >= 3 ? Math.min(window.devicePixelRatio || 1, 2) : 1;
  renderer.setPixelRatio(pr); renderer.setSize(ancho, alto, false);
  renderer.domElement.style.width = '100%'; renderer.domElement.style.height = '100%';
  composer.setPixelRatio(pr); composer.setSize(ancho, alto);
  camera.aspect = ancho / alto; camera.updateProjectionMatrix();
  invalidar();
}
new ResizeObserver(() => ajustarTam()).observe(vista);

let pendiente = true;
function invalidar() { pendiente = true; }
controls.addEventListener('change', invalidar);
function dibujar() {
  pBokeh.uniforms.focus.value = camera.position.distanceTo(controls.target);
  composer.render();
}
function bucle(ahora) {
  requestAnimationFrame(bucle);
  let mueve = false;
  if (viaje) {
    const t = Math.min(1, (ahora - viaje.ini) / viaje.dur), e = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
    camera.position.lerpVectors(viaje.p0, viaje.p1, e); controls.target.lerpVectors(viaje.t0, viaje.t1, e);
    if (t >= 1) viaje = null;
    mueve = true;
  }
  if (est.giro && actual) { actual.g.rotation.y += 0.006; mueve = true; }
  const amort = controls.update();
  if (mueve || amort || pendiente) {
    pendiente = false;
    dibujar();
    if (mueve) medir(ahora); else ultimo = 0;
  }
}

// ---------- Exportar PNG ----------
async function exportar(escala, transparente) {
  const max = renderer.capabilities.maxTextureSize || 4096;
  let w = Math.round(ancho * escala), h = Math.round(alto * escala);
  const tope = Math.min(4096, max), k = Math.min(1, tope / Math.max(w, h));
  w = Math.round(w * k); h = Math.round(h * k);
  const prPrev = renderer.getPixelRatio(), bg = scene.background;
  renderer.setPixelRatio(1); renderer.setSize(w, h, false);
  let blob;
  try {
    if (transparente) {
      scene.background = null; renderer.setClearColor(0x000000, 0);
      renderer.render(scene, camera);
    } else {
      composer.setPixelRatio(1); composer.setSize(w, h);
      dibujar();
    }
    blob = await new Promise((ok) => renderer.domElement.toBlob(ok, 'image/png'));
  } finally {
    scene.background = bg;
    renderer.setPixelRatio(prPrev);
    ajustarTam();
    dibujar();
  }
  return { blob, w, h };
}

// ---------- Controles de la página ----------
function marcar(grupo, valor) {
  document.querySelectorAll(`[data-g="${grupo}"] button`).forEach((b) => b.setAttribute('aria-pressed', String(b.dataset.v === valor)));
}
function elegirPlantilla(p) {
  const P = PLANTILLAS[p];
  est.plantilla = p; est.color = P.color; est.material = P.material;
  $('color').value = P.color;
  $('marca').placeholder = P.marca; $('linea').placeholder = P.linea || 'Sin línea secundaria';
  $('linea').disabled = p === 'logo';
  marcar('plantilla', p); marcar('material', est.material);
  reconstruir(); irCamara(est.camara, true);
}
document.querySelectorAll('[data-g] button').forEach((b) => {
  b.type = 'button';
  b.addEventListener('click', () => {
    const g = b.parentElement.dataset.g, v = b.dataset.v;
    if (g === 'plantilla') return elegirPlantilla(v);
    est[g] = v; marcar(g, v);
    if (g === 'material') reconstruir();
    else if (g === 'fondo') { aplicarFondo(); invalidar(); }
    else if (g === 'luz') aplicarLuz();
    else if (g === 'camara') irCamara(v);
    else if (g === 'calidad') { if (v !== 'auto') aplicarNivel(+v); else aplicarNivel(3); }
  });
});
let tTexto = 0;
$('color').addEventListener('input', (e) => { est.color = e.target.value; clearTimeout(tTexto); tTexto = setTimeout(reconstruir, 60); });
$('colorMarca').addEventListener('input', (e) => { est.colorMarca = e.target.value; aplicarFondo(); invalidar(); });
for (const id of ['marca', 'linea']) $(id).addEventListener('input', (e) => { est[id] = e.target.value; clearTimeout(tTexto); tTexto = setTimeout(reconstruir, 250); });
$('desenfoque').addEventListener('change', (e) => { est.desenfoque = e.target.checked; pBokeh.enabled = est.desenfoque && est.nivel >= 2; invalidar(); });
$('giro').addEventListener('change', (e) => { est.giro = e.target.checked; if (!est.giro && actual) { actual.g.rotation.y = 0; invalidar(); } });
$('imagen').addEventListener('change', (e) => {
  const f = e.target.files && e.target.files[0];
  if (!f) return;
  if (f.size > 8 * 1024 * 1024) { avisar('La imagen pesa más de 8 MB: usa una más liviana.'); return; }
  const lector = new FileReader();
  lector.onload = () => {
    const im = new Image();
    im.onload = () => { est.imagen = im; $('quitar').hidden = false; reconstruir(); avisar('Etiqueta cambiada por tu imagen.'); };
    im.onerror = () => avisar('No pude leer esa imagen: prueba con PNG, JPG o WebP.');
    im.src = lector.result;
  };
  lector.readAsDataURL(f);
});
$('quitar').addEventListener('click', () => { est.imagen = null; $('imagen').value = ''; $('quitar').hidden = true; reconstruir(); });
function avisar(t) { const a = $('aviso'); a.textContent = t; clearTimeout(avisar.t); avisar.t = setTimeout(() => { a.textContent = ''; }, 4000); }
$('descargar').addEventListener('click', async () => {
  const b = $('descargar'); b.disabled = true; const txt = b.textContent; b.textContent = 'Renderizando…';
  try {
    const escala = +document.querySelector('[data-g="escala"] [aria-pressed="true"]').dataset.v;
    const { blob, w, h } = await exportar(escala, $('transparente').checked);
    if (!blob) throw new Error('sin imagen');
    const url = URL.createObjectURL(blob), a = document.createElement('a');
    a.href = url; a.download = `estudio3d-${est.plantilla}-${w}x${h}.png`; document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 4000);
    avisar(`Descargado: ${w} × ${h} px.`);
  } catch (err) {
    avisar('No se pudo exportar a ese tamaño: prueba con 2x.');
  } finally { b.disabled = false; b.textContent = txt; }
});
document.querySelectorAll('[data-g="escala"] button').forEach((b) => b.addEventListener('click', () => marcar('escala', b.dataset.v)));

// API para pruebas y para el video de la clase
window.estudio3d = {
  async set(op) {
    Object.assign(est, op);
    if (op.plantilla) { const P = PLANTILLAS[op.plantilla]; if (!op.color) est.color = P.color; if (!op.material) est.material = P.material; $('color').value = est.color; marcar('plantilla', op.plantilla); marcar('material', est.material); }
    if (op.calidad) aplicarNivel(op.calidad === 'auto' ? 3 : +op.calidad);
    aplicarLuz(); reconstruir(); irCamara(est.camara, true);
    pBokeh.enabled = est.desenfoque && est.nivel >= 2;
    dibujar();
    return { nivel: est.nivel, w: ancho, h: alto };
  },
  png() { dibujar(); return renderer.domElement.toDataURL('image/png'); },
  async exportar(escala, transparente) { const r = await exportar(escala, transparente); return { w: r.w, h: r.h, bytes: r.blob ? r.blob.size : 0 }; },
  estado: () => ({ ...est, imagen: !!est.imagen }),
};

// Prueba de rendimiento al abrir: 24 cuadros girando; si el equipo es lento, baja AO y bloom solo.
(function prueba() {
  if (est.calidad !== 'auto') return;
  let n = 0, t0 = 0;
  const paso = (t) => {
    if (n === 4) t0 = t;
    if (n < 28) { n++; composer.render(); requestAnimationFrame(paso); return; }
    const prom = (t - t0) / 24;
    if (prom > 45) aplicarNivel(1); else if (prom > 26) aplicarNivel(2);
    invalidar();
  };
  requestAnimationFrame(paso);
})();
document.documentElement.dataset.listo = '1';
