# extractor.py · scraping ético en el servidor (curso Web scraping ético con IA, JMD Learning)
# Sitio de práctica: quotes.toscrape.com, hecho para practicar scraping.
# Sesión con cookie, token CSRF, paginación, pausas, reintentos y CSV/JSON limpios.
import csv, json, os, random, re, time
from urllib import robotparser

try:
    import httpx as http          # ya viene en el servidor del curso
except ImportError:
    import requests as http       # en tu máquina: pip install requests

BASE = "https://quotes.toscrape.com"
PAUSA = float(os.environ.get("PAUSA", "1.5"))         # segundos entre pedidos
MAX_PAGINAS = int(os.environ.get("MAX_PAGINAS", "5"))
UA = "CursoJMD-scraping-etico/1.0 (practica; " + os.environ.get("CONTACTO", "sin-contacto") + ")"
CABECERAS = {"User-Agent": UA}
if os.environ.get("TOKEN"):        # el token viaja en la cabecera y vive en una variable, nunca en el código
    CABECERAS["Authorization"] = "Bearer " + os.environ["TOKEN"]


def nueva_sesion():
    if http.__name__ == "httpx":
        return http.Client(headers=CABECERAS, follow_redirects=True)
    s = http.Session()
    s.headers.update(CABECERAS)
    return s


def pedir(s, metodo, url, **kw):
    for intento in range(1, 5):
        r = s.request(metodo, url, timeout=20, **kw)
        if r.status_code in (429, 500, 502, 503, 504):
            try:
                espera = int(r.headers.get("Retry-After", ""))
            except ValueError:
                espera = 2 ** intento                    # 2, 4, 8, 16 s
            print(f"  {r.status_code} en {url}: espero {espera} s (intento {intento})")
            time.sleep(espera)
            continue
        r.raise_for_status()
        time.sleep(PAUSA + random.uniform(0, 0.5))      # ritmo humano, con un poco de azar
        return r
    raise SystemExit(f"{url} no respondió tras 4 intentos: me detengo")


def limpiar(t):
    return re.sub(r"\s+", " ", t).strip().strip("“”\"").strip()


# 1. Reglas primero: robots.txt
rp = robotparser.RobotFileParser(BASE + "/robots.txt")
rp.read()
if not rp.can_fetch(UA, BASE + "/api/quotes"):
    raise SystemExit("robots.txt no permite esta ruta: me detengo")
PAUSA = max(PAUSA, rp.crawl_delay(UA) or 0)
print(f"robots.txt OK · pausa de {PAUSA} s entre pedidos")

# 2. Sesión: la cookie la guarda la sesión; el token CSRF sale del formulario
s = nueva_sesion()
html = pedir(s, "GET", BASE + "/login").text
token = re.search(r'name="csrf_token" value="([^"]+)"', html).group(1)
pedir(s, "POST", BASE + "/login", data={
    "csrf_token": token,
    "username": os.environ.get("USUARIO", "alumno"),
    "password": os.environ.get("CLAVE", "practica"),
})
dentro = "Logout" in pedir(s, "GET", BASE + "/").text

# 3. Paginación por la API del sitio (si hay API, úsala antes que el HTML)
filas, vistas, pagina = [], set(), 1
while pagina <= MAX_PAGINAS:
    datos = pedir(s, "GET", BASE + "/api/quotes", params={"page": pagina}).json()
    for q in datos["quotes"]:
        fila = {"frase": limpiar(q["text"]), "autor": limpiar(q["author"]["name"]),
                "etiquetas": ";".join(q["tags"]), "pagina": pagina}
        clave = (fila["frase"].lower(), fila["autor"].lower())
        if clave not in vistas:                          # sin filas repetidas
            vistas.add(clave)
            filas.append(fila)
    print(f"página {pagina}: {len(filas)} filas")
    if not datos.get("has_next"):
        break
    pagina += 1

# 4. Exportar limpio: CSV que Excel abre con tildes (utf-8-sig) y JSON legible
with open("frases.csv", "w", newline="", encoding="utf-8-sig") as f:
    w = csv.DictWriter(f, fieldnames=["frase", "autor", "etiquetas", "pagina"])
    w.writeheader()
    w.writerows(filas)
with open("frases.json", "w", encoding="utf-8") as f:
    json.dump(filas, f, ensure_ascii=False, indent=2)
print(f"Listo: {len(filas)} filas en frases.csv y frases.json")
EOF

Lee tu script en cuatro bloques
  1. 1 · Reglas: robotparser lee el robots.txt; si la ruta está prohibida, el script se detiene solo.
  2. 2 · Sesión: saca el csrf_token del formulario, inicia sesión y la cookie queda guardada en la sesión.
  3. 3 · Páginas: pide /api/quotes?page=N mientras has_next sea verdadero, con pausa y reintentos.
  4. 4 · Exportar: limpia comillas y espacios, quita repetidas y escribe CSV y JSON.
